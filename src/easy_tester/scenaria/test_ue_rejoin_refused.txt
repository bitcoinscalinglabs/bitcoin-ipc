# A key joins a subnet once: after its collateral matures and a checkpoint rotates it out,
# a second join-ue of the same key is refused, and so is a second join-ue of a member.
# Run with testers/db-ue.txt or testers/monitor-ue.txt. `block N` means what follows goes
# into block N, so a removal at T - 1 is visible at `block T`.

setup

validator1
validator2
validator3
validator4

subnet1
whitelist validator1 validator2 validator3 validator4
min 4

scenario

block 105
create subnet1

block 110
join-ue subnet1 validator1 100_000 130
join-ue subnet1 validator2 100_000 160
join-ue subnet1 validator3 100_000 170
join-ue subnet1 validator4 100_000 180

block 112
read validators subnet1
expect result.count = 4
expect result.waiting = false
# the multisig needs funds before a checkpoint can be built
deposit subnet1 validator1 500_000_000

# a member cannot join again
block 115
join-ue subnet1 validator2 100_000 400

block 117
read ue_collaterals subnet1
expect result.count = 4
expect result.validator2.count = 1

# state after block 129 = T - 1: validator1 left the next committee
block 130
read validators subnet1
expect result.names = validator2,validator3,validator4
expect result.waiting = true

# the checkpoint after the removal rotates validator1 out
checkpoint subnet1

block 133
read validators subnet1 current
expect result.names = validator2,validator3,validator4
expect result.waiting = false

# the key is free of the committee, and still cannot join again
join-ue subnet1 validator1 120_000 200

block 135
read validators subnet1
expect result.names = validator2,validator3,validator4
expect result.waiting = false
read ue_collaterals subnet1
expect result.count = 4
expect result.validator1.count = 1
expect result.validator1.removed = true
read withdrawals subnet1
expect result.count = 0
