# join-ue requests that are rejected, and the ones that are accepted right after.

setup

validator1
validator2
validator3
validator4
validator5

subnet1
whitelist validator1 validator2 validator3 validator5
min 4

scenario

block 105
create subnet1

# not whitelisted before bootstrap
block 106
join-ue subnet1 validator4 100_000 200

block 108
read ue_collaterals subnet1
expect result.count = 0

# below the minimum collateral, and a timeout below the fourth epoch from the join (140)
block 110
join-ue subnet1 validator1 99_999 200
join-ue subnet1 validator2 100_000 140

block 112
read ue_collaterals subnet1
expect result.count = 0

# exactly the minimum collateral is accepted, and a timeout in the fourth epoch from the
# join (144 at block 113); the subnet bootstraps
block 113
join-ue subnet1 validator1 100_000 200
join-ue subnet1 validator2 100_000 144
join-ue subnet1 validator3 100_000 200
join-ue subnet1 validator5 100_000 200

block 115
read validators subnet1
expect result.count = 4
expect result.names = validator1,validator2,validator3,validator5
read ue_collaterals subnet1
expect result.validator2.timeout = 144
# the same key again, with another timeout: rejected, the first record stays
join-ue subnet1 validator1 100_000 300

block 117
read validators subnet1
expect result.count = 4
read ue_collaterals subnet1
expect result.count = 4
expect result.validator1.count = 1
expect result.validator1.timeout = 200

# epoch 0: nobody joins before the first checkpoint
block 118
join-ue subnet1 validator4 100_000 300

block 120
read ue_collaterals subnet1
expect result.count = 4
deposit subnet1 validator1 500_000_000
presign subnet1 validator1 by validator1 validator2 validator3
presign subnet1 validator2 by validator1 validator2 validator3
presign subnet1 validator3 by validator1 validator2 validator3
presign subnet1 validator5 by validator1 validator2 validator3

# the first checkpoint, H_0 = 121: C_1 needs T > 151, so validator2 (144) is not in it and
# leaves with the one unit of value C_gen gave it
block 121
checkpoint subnet1

block 122
read subnet_state subnet1
expect result.activated = true
expect result.h_0 = 121
expect result.epoch = 1
expect result.last_rotation_epoch = 0
# C_1 waits for the rotation checkpoint of epoch 1; C_gen is current
read validators subnet1
expect result.names = validator1,validator3,validator5
expect result.waiting = true
read validators subnet1 current
expect result.names = validator1,validator2,validator3,validator5
expect result.waiting = true
read withdrawals subnet1
expect result.count = 1
expect result.validator2 = 10_000
# after activation anyone can join, with a timeout in epochs 5 to 1004 from H_0 in epoch 1:
# 161 is rejected, 162 accepted
join-ue subnet1 validator4 100_000 161
join-ue subnet1 validator4 100_000 162

block 124
read validators subnet1
expect result.count = 3
expect result.waiting = true
read validators subnet1 current
expect result.count = 4
read ue_collaterals subnet1
expect result.validator4.count = 1
expect result.validator4.timeout = 162
presign subnet1 validator4 by validator1 validator3 validator5

# the next checkpoint after H_0, the rotation checkpoint of epoch 1: it applies C_1 and
# confirms the withdrawal of validator2
block 125
checkpoint subnet1

block 126
read subnet_state subnet1
expect result.last_rotation_epoch = 1
read validators subnet1 current
expect result.names = validator1,validator3,validator5
expect result.waiting = false
read withdrawals subnet1
expect result.count = 0
read ue_collaterals subnet1
expect result.validator4.slashable = true

# the cutoff at 131 computes C_2 with boundary 161: validator4 (162) is seated
block 132
read validators subnet1
expect result.names = validator1,validator3,validator4,validator5
expect result.waiting = true
read withdrawals subnet1
expect result.count = 0

# the cutoff at 141 computes C_3 with boundary 171: validator4 leaves with its collateral
block 142
read validators subnet1
expect result.names = validator1,validator3,validator5
read withdrawals subnet1
expect result.count = 1
expect result.validator4 = 100_000
