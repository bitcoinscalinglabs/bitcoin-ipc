# Unilateral exit end to end: a dropped collateral, a refused second join and unstake-ue spending the matured collateral.

setup

validator1
validator2
validator3
validator4

subnet1
whitelist validator1 validator2 validator3 validator4
min 4

scenario

block 105
create subnet1

block 110
join-ue subnet1 validator1 100_000 165
join-ue subnet1 validator2 100_000 300
join-ue subnet1 validator3 100_000 300
join-ue subnet1 validator4 100_000 400

block 112
read validators subnet1
expect result.count = 4
expect result.waiting = false
deposit subnet1 validator1 500_000_000
presign subnet1 validator1 by validator1 validator2 validator3
presign subnet1 validator2 by validator1 validator2 validator3
presign subnet1 validator3 by validator1 validator2 validator3
presign subnet1 validator4 by validator1 validator2 validator3

# the first checkpoint: H_0 = 114
block 114
checkpoint subnet1

# before maturity the provider refuses: nothing to spend
block 120
unstake-ue subnet1 validator1

block 122
read ue_collaterals subnet1
expect result.spent_count = 0
expect result.validator1.spent = false

# the cutoff at 144 (boundary 174) drops validator1 (165)
block 145
read validators subnet1
expect result.names = validator2,validator3,validator4
expect result.waiting = true
read withdrawals subnet1
expect result.count = 1
expect result.validator1 = 100_000

# at tip 165 = T the exit is accepted; block 166 mines it
block 166
unstake-ue subnet1 validator1

block 168
read ue_collaterals subnet1
expect result.spent_count = 1
expect result.validator1.spent = true
# the rotation checkpoint adopts the configuration without validator1; nothing is paid from
# the multisig
checkpoint subnet1

block 171
read validators subnet1 current
expect result.names = validator2,validator3,validator4
expect result.waiting = false
read withdrawals subnet1
expect result.count = 0

# a key joins a subnet once: the second join-ue is refused even after the rotation
join-ue subnet1 validator1 120_000 400

block 173
read validators subnet1
expect result.names = validator2,validator3,validator4
expect result.waiting = false
read ue_collaterals subnet1
expect result.validator1.count = 1
expect result.validator1.spent = true
expect result.spent_count = 1

# a further call spends nothing: the only collateral is already spent
unstake-ue subnet1 validator1

block 175
read ue_collaterals subnet1
expect result.spent_count = 1
