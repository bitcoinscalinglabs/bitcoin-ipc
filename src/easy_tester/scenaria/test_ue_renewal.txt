# A waiting validator cannot renew its key with a second join-ue in the block its collateral
# matures: a key joins a subnet once. The monitor removes the old membership before the
# block's transactions and refuses the new join-ue.
# Run with testers/db-ue.txt or testers/monitor-ue.txt.

setup

validator1
validator2
validator3
validator4
validator5

subnet1
whitelist validator1 validator2 validator3 validator4
min 4

scenario

block 105
create subnet1

block 110
join-ue subnet1 validator1 100_000 400
join-ue subnet1 validator2 100_000 400
join-ue subnet1 validator3 100_000 400
join-ue subnet1 validator4 100_000 400

block 112
join-ue subnet1 validator5 100_000 130

block 114
read validators subnet1
expect result.count = 5
expect result.waiting = true

# block 129 = T - 1 of validator5: it leaves the waiting committee; its new join-ue is refused
block 129
join-ue subnet1 validator5 100_000 300

block 130
read validators subnet1
expect result.count = 4
expect result.names = validator1,validator2,validator3,validator4
read ue_collaterals subnet1
expect result.validator5.count = 1
expect result.validator5.timeout = 130
expect result.validator5.removed = true
read withdrawals subnet1
expect result.count = 1
expect result.validator5 = 100_000
