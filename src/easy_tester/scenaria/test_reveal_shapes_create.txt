# The monitor reads a create only from a reveal with exactly one input and one output.

setup
validator1
validator2
validator3
validator4
subnet_accepted
whitelist validator1 validator2 validator3 validator4
min 4
subnet_two_inputs
whitelist validator1 validator2 validator3 validator4
min 4
subnet_data_second
whitelist validator1 validator2 validator3 validator4
min 4
subnet_two_outputs
whitelist validator1 validator2 validator3 validator4
min 4
subnet_other_output
whitelist validator1 validator2 validator3 validator4
min 4

scenario

block 102
create subnet_accepted
create subnet_two_inputs @input-after
create subnet_data_second @input-before
create subnet_two_outputs @extra-output
create subnet_other_output @other-output

block 103
read subnet_genesis subnet_accepted
expect result.exists = true
expect result.bootstrapped = false
expect result.min_validators = 4
read subnet_genesis subnet_two_inputs
expect result.exists = false
read subnet_genesis subnet_data_second
expect result.exists = false
read subnet_genesis subnet_two_outputs
expect result.exists = false
read subnet_genesis subnet_other_output
expect result.exists = true

# The subnet_accepted subnet works
join subnet_accepted validator1 100_000_000
join subnet_accepted validator2 100_000_000
join subnet_accepted validator3 100_000_000
join subnet_accepted validator4 100_000_000

block 105
read subnet_genesis subnet_accepted
expect result.bootstrapped = true
read validators subnet_accepted
expect result.count = 4
